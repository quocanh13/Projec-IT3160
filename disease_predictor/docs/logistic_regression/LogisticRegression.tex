\noindent Đầu vào: $x_1, x_2, \ldots, x_n$ \\
Trọng số: $w_1, w_2, \ldots, w_n$ \\
Bias: $b$

\vspace{0.3cm}
\noindent Đặt: \[z = w_1x_1 + w_2x_2 + \dots + w_nx_n + b = \sum_{i = 1}^{n}w_ix_i + b\] 

\noindent Hàm sigmoid: \[\sigma(x) = \frac{1}{1+e^{-x}}\]
Hàm sigmoid là hàm đồng biến trên $(-\infty, +\infty)$ và có miền giá trị là $(0, 1)$.

\vspace{0.3cm}
\noindent Gọi đầu ra là $\hat{y}$: \[\hat{y} = \sigma(z) = \frac{1}{1+e^{-z}}\]
Đặt nhãn thực tế là: \[y \in \{0, 1\}\]
\section{Mean Squared Error}
Gọi hàm loss (MSE) là: \[h(\hat{y}) = (\hat{y} - y)^2\] 

\noindent Đạo hàm của hàm loss theo trọng số $w_i$ là:
\[
    \frac{\partial h}{\partial w_i} = 
    \frac{\partial h}{\partial \hat{y}} \cdot
    \frac{\partial \hat{y}}{\partial z} \cdot
    \frac{\partial z}{\partial w_i} =
    2(\hat{y} - y) \cdot \hat{y}(1 - \hat{y}) \cdot x_i = 
    2\hat{y}(\hat{y} - y)(1 - \hat{y})x_i
\]

\noindent Đạo hàm của hàm loss theo bias $b$ là:
\[
    \frac{\partial h}{\partial b} = 
    \frac{\partial h}{\partial \hat{y}} \cdot
    \frac{\partial \hat{y}}{\partial z} \cdot
    \frac{\partial z}{\partial b} =
    2(\hat{y} - y) \cdot \hat{y}(1 - \hat{y}) \cdot 1 = 
    2\hat{y}(\hat{y} - y)(1 - \hat{y})
\]

\noindent Đặt :
\[\delta = 2(\hat{y} - y) \cdot \hat{y}(1 - \hat{y})\]

\noindent Ta có :
\[
    \begin{cases}
        \frac{\partial h}{\partial w_i} = \delta x_i \\
        \frac{\partial h}{\partial b} = \delta
    \end{cases}
\]

\noindent Biểu diễn dạng ma trận :

\vspace{0.3cm}
\noindent Gọi m là số mẫu - sample, n là số phần tử trong một mẫu
\[
    X = \begin{bmatrix}
        x_{1, 1} & x_{1, 2} & \hdots & x_{1, n} \\
        x_{2, 1} & x_{2, 2} & \hdots & x_{2, n} \\
        \hdots & \hdots & \hdots & \hdots \\
        x_{m, 1} & x_{m, 2} & \hdots & x_{m, n} \\
    \end{bmatrix} (m \times n)
\]

\[
    W = \begin{bmatrix}
        w_1 \\ w_2 \\ \vdots \\ w_n
    \end{bmatrix} (n \times 1)
\]

\[
    B = \begin{bmatrix}
        b \\ b \\ \vdots \\ b
    \end{bmatrix} (m \times 1)
\]

\[
    Z = XW + B \quad (m \times 1)
\]

\[
    \hat{Y} = \sigma(Z) \quad (m \times 1)
\]

\[
    Y = \begin{bmatrix}
        y_1 \\ y_2 \\ \vdots \\ y_m
    \end{bmatrix} (m \times 1)
\]

\[
    \delta = \frac{2}{m}(\hat{Y} - Y) \odot \hat{Y} \odot (1-\hat{Y}) \quad (m \times 1)
\]

\noindent Lý do có chia cho m là do công thức delta ở phần biểu diễn vô hướng chỉ là 1 mẫu còn ở đây là m mẫu nên ta sẽ lấy trung bình trên m mẫu đó

\vspace{0.3cm}
\noindent Đặt :
\[
    I_m = \begin{bmatrix}
        1 \\ 1 \\ \vdots \\ 1
    \end{bmatrix} (m \times 1)
\]

\[
    \begin{cases}
        \frac{\partial h}{\partial W} = X^T \cdot \delta \quad (n \times 1) \\
        \frac{\partial h}{\partial B} = I_m^T \cdot \delta \quad (1 \times 1)
    \end{cases}
\]
