\begin{center}
    {\LARGE SOFTMAX LOGISTIC REGRESSION}
\end{center}

\noindent Đầu vào: $x_1, x_2, \ldots, x_n$ \\
Trọng số: 
\[
    \begin{matrix}
        w_{1, 1} & w_{1, 2} & \ldots & w_{1, n} \\
        w_{2, 1} & w_{2, 2} & \ldots & w_{2, n} \\
        \ldots & \ldots & \ldots & \ldots \\
        w_{m, 1} & w_{m, 2} & \ldots & w_{m, n}
    \end{matrix}
\] 

\noindent m là số lượng lớp (class) dự đoán, n là kích thước đầu vào

\noindent Bias: $b_1, b_2, \ldots b_m$

\vspace{0.3cm}
\noindent Đặt: 
\[
z_i = w_{i, 1}x_1 + w_{i, 2}x_2 + \dots + w_{i, n}x_n + b_i = 
    \sum_{k = 1}^{n}w_{i, k}x_k + b_i
    \quad (i \leq m)
\] 

\noindent Hàm softmax tại i với đầu vào là chuỗi z:
\[
    \sigma_i(z) = \frac{e^{z_i}}{\sum_{k = 1}^{m}e^{z_k}}
\]

\vspace{0.3cm}
\noindent Gọi đầu ra là:
\[
    \hat{y_i} = \sigma_i(z) = \frac{e^{z_i}}{\sum_{k = 1}^{m}e^{z_k}} 
    \quad (i \leq m)
\]
Đặt nhãn thực tế là: \[y_1, y_2, \ldots, y_m \qquad y_i \in \{0; 1\}\]

\vspace{0.3cm}
\noindent Hàm loss (Cross Entropy Loss):
\[
    L = -\left(y_1\ln(\hat{y_1}) + y_2\ln(\hat{y_2}) + \ldots + y_m\ln(\hat{y_m})\right) = 
     -\sum_{i = 1}^{m}y_i\ln(\hat{y_i})
\]

\noindent Đạo hàm của hàm Loss theo trọng số là:
\[
    \frac{\partial L}{\partial w_{i, j}} = 
    (\hat{y_i} - y_i)x_j
\]

\noindent Đạo hàm của hàm Loss theo bias là:
\[
    \frac{\partial L}{\partial b_i} = 
    \hat{y_i} - y_i
\]

\noindent Đặt :
\[
    \delta_i = \hat{y_i} - y_i
\]

\noindent Ta được:
\[
    \begin{cases}
    \frac{\partial L}{\partial w_{i, j}} = 
    \delta_i x_j \\
    \frac{\partial L}{\partial b_i} = \delta_i
    \end{cases}
\]

\noindent Biểu diễn dạng ma trận :

\vspace{0.3cm}
\noindent Gọi m là số lớp dự đoán, n là kích thước đầu vào, q là số mẫu
\[
    X = \begin{bmatrix}
        x_{1, 1} & x_{1, 2} & \hdots & x_{1, n} \\
        x_{2, 1} & x_{2, 2} & \hdots & x_{2, n} \\
        \hdots & \hdots & \hdots & \hdots \\
        x_{q, 1} & x_{q, 2} & \hdots & x_{q, n} \\
    \end{bmatrix} (q \times n)
\]

\[
    W = \begin{bmatrix}
        w_{1, 1} & w_{1, 2} & \hdots & w_{1, n} \\
        w_{2, 1} & w_{2, 2} & \hdots & w_{2, n} \\
        \hdots & \hdots & \hdots & \hdots \\
        w_{m, 1} & w_{m, 2} & \hdots & w_{m, n} \\
    \end{bmatrix} (m \times n)
\]

\[
    B = \begin{bmatrix}
        b_1 \\ b_2 \\ \vdots \\ b_m
    \end{bmatrix} (m \times 1)
\]

\[
    Z = XW^T + B \quad (q \times m)
\]


\vspace{0.2cm}

\noindent Do B có kích thước $m \times 1$ nên ta sẽ coi khi cộng $B$ tự động mở rộng lên thành $m \times q$ với các cột chính là sao chép từ 1 cột gốc ban đầu
\[
    \hat{Y} = \sigma(Z) \quad (q \times m)
\]
\[
    Y = \begin{bmatrix}
        y_{1, 1} & y_{1, 2} & \vdots & y_{1, m} \\
        y_{2, 1} & y_{2, 2} & \vdots & y_{2, m} \\
        \hdots & \hdots & \vdots & \hdots \\
        y_{q, 1} & y_{q, 2} & \vdots & y_{q, m}
    \end{bmatrix} (q \times m)
\]

\[
    \delta = \frac{1}{q}(\hat{Y} - Y) \quad (q \times m)
\]

\noindent Đặt :
\[
I_q = \begin{bmatrix}
    1 \\ 1 \\ \vdots \\ 1    
\end{bmatrix} (q \times 1)
\]

\[
    \begin{cases}
        \frac{\partial h}{\partial W} = \delta^T \cdot X \quad (m \times n) \\
        \frac{\partial h}{\partial B} = \delta^T \cdot I_q \quad (m \times 1)
    \end{cases}
\]
